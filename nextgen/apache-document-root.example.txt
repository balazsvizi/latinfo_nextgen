# Apache / Laragon: DocumentRoot = a projekt gyökér könyvtára (ahol a nextgen/ és lanueva/ mappák vannak).
# Helyi Laragon: a gyökér .htaccess (gitignore) tartalmazza a szabályokat.
# Production (WordPress): illeszd be az apache-wordpress-public-routes.example.txt tartalmát
# a public_html/.htaccess „BEGIN WordPress” blokkja ELÉ.

DirectoryIndex index.php

RewriteEngine On
# Gyökér → backoffice (ne DirectoryIndex nextgen/index.php — almappákban 403)
RewriteRule ^$ nextgen/index.php [L]
RewriteRule ^lanueva$ /lanueva/ [R=301,L]
RewriteRule ^feedback$ /feedback/ [R=301,L]
# Régi URL-ek
RewriteRule ^visszajelzes/?$ /feedback/ [R=301,L,QSA]
RewriteRule ^visszajelzés/?$ /feedback/ [R=301,L,NE,QSA]
